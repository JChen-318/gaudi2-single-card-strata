# Compile the original TPC kernel and embed its ELF in the glue library. No SDK
# header or binary is vendored. The only shipped device code is built from ours.
set(tpc_blobs)
foreach(kernel IN ITEMS gdn_step gdn_conv gdn_prepare gdn_norm row_norm hc_mix hc_write hc_mix_window hc_write_window ple_gate ple_conv packed_matmul activation_quant row_norm_simd fp8_encode fp8_gemv gdn_prepare_simd gdn_norm_simd gdn_step_bcast fp8_gemv_hilo fp8_gemv_colpair gdn_step_cached iq4q8_gemv q8_quant route_topk q8_quant_shuffle qsa_prep route_share iq4q8_gemv_rows gdn_step_pipe gdn_conv_rows gdn_prepare_rows gdn_norm_rows iq4q8_gemv_pipe q8_quant_packed iq4q8_gemv_packed iq4q8_gemv_tile fp8_encode_stacked fp8_linear_finish moe_act_packed row_norm_dual hc_mix_window_dual gdn_norm_rows_bf16 hc_write_moe gdn_step_rows4 fp8_encode_fixed gdn_conv_rows8 qsa_gather gdn_conv_final gdn_step_final qsa_virtual gdn_step_window route_group_starts route_group_order iq4q8_gemv_group iq4q8_gemv_packed_pf u4_moe_bucket u4_moe_gather u4_moe_gu_epi u4_moe_down_epi u4_moe_join u4_route_blocks u4_bucket_blocks fp8_batch_encode fp8_batch_finish bw_read_probe route_group_blocks iq4q8_gemv_pair iq4_expand_bf16 moe_mme_join moe_mme_act moe_mme_join_perm moe_mme_count moe_mme_prefix moe_mme_scatter moe_mme_gather moe_mme_tilemap iq4_expand_tiles ple_gate_simd gdn_norm_rows8 gdn_norm_fast iq4_expand_tiles_fp8 moe_mme_gather_fp8 moe_mme_act_fp8 moe_mme_join_fp8 hc_write_moe_routed qsa_index_prep qsa_index_cells qsa_select_mask qsa_select_expand moe_mme_place qsa_gather_fp8 qsa_decode_blocks qsa_gather_blocks_fp8 qsa_window_blocks qsa_gather_window_fp8 hc_down_bf16 hc_mix_dual gemv_bf16 ple_gate_rows ple_conv_rows qsa_select_expand_fold qsa_gather_rows_fp8 qsa_select_bits hc_down_rows_bf16 qsa_gather_window_bf16 qsa_verify_softmax qsa_verify_partials)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${kernel}.o"
    COMMAND "${STRATA_TPC_COMPILER}" -march=gaudi2 -O2 -c
      "${CMAKE_CURRENT_SOURCE_DIR}/tpc/${kernel}.c" -o "${CMAKE_CURRENT_BINARY_DIR}/${kernel}.o"
    DEPENDS "tpc/${kernel}.c" tpc/gdn_math.h tpc/moe_mme_fp8.h tpc/qsa_select_mask.c VERBATIM)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${kernel}_blob.S" CONTENT
    ".section .rodata\n.global strata_${kernel}_elf_start\n.hidden strata_${kernel}_elf_start\n.global strata_${kernel}_elf_end\n.hidden strata_${kernel}_elf_end\nstrata_${kernel}_elf_start:\n.incbin \"${CMAKE_CURRENT_BINARY_DIR}/${kernel}.o\"\nstrata_${kernel}_elf_end:\n.section .note.GNU-stack,\"\",@progbits\n")
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${kernel}_blob.o"
    COMMAND "${CMAKE_CXX_COMPILER}" -c "${CMAKE_CURRENT_BINARY_DIR}/${kernel}_blob.S"
      -o "${CMAKE_CURRENT_BINARY_DIR}/${kernel}_blob.o"
    DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/${kernel}.o" "${CMAKE_CURRENT_BINARY_DIR}/${kernel}_blob.S" VERBATIM)
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/${kernel}_blob.o" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
  list(APPEND tpc_blobs "${CMAKE_CURRENT_BINARY_DIR}/${kernel}_blob.o")
endforeach()
# Fast-math twins (STRATA_HPU_FAST_SIGMOID): the same sources built with -DSTRATA_FAST_MATH (tpc/gdn_math.h).
foreach(kernel IN ITEMS hc_mix_window_dual gdn_conv_final moe_mme_act row_norm_dual)
  set(twin ${kernel}_fast)
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${twin}.o"
    COMMAND "${STRATA_TPC_COMPILER}" -march=gaudi2 -O2 -DSTRATA_FAST_MATH -c
      "${CMAKE_CURRENT_SOURCE_DIR}/tpc/${kernel}.c" -o "${CMAKE_CURRENT_BINARY_DIR}/${twin}.o"
    DEPENDS "tpc/${kernel}.c" tpc/gdn_math.h VERBATIM)
  file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${twin}_blob.S" CONTENT
    ".section .rodata\n.global strata_${twin}_elf_start\n.hidden strata_${twin}_elf_start\n.global strata_${twin}_elf_end\n.hidden strata_${twin}_elf_end\nstrata_${twin}_elf_start:\n.incbin \"${CMAKE_CURRENT_BINARY_DIR}/${twin}.o\"\nstrata_${twin}_elf_end:\n.section .note.GNU-stack,\"\",@progbits\n")
  add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/${twin}_blob.o"
    COMMAND "${CMAKE_CXX_COMPILER}" -c "${CMAKE_CURRENT_BINARY_DIR}/${twin}_blob.S"
      -o "${CMAKE_CURRENT_BINARY_DIR}/${twin}_blob.o"
    DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/${twin}.o" "${CMAKE_CURRENT_BINARY_DIR}/${twin}_blob.S" VERBATIM)
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/${twin}_blob.o" PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
  list(APPEND tpc_blobs "${CMAKE_CURRENT_BINARY_DIR}/${twin}_blob.o")
endforeach()
